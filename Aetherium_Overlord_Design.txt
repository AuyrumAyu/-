// ===== capability/MagicPath.java =====
package com.aetherium.overlord.capability;

public enum MagicPath {
    NONE("Без пути", 0xFFFFFF),
    PYROMANCER("Пиромант", 0xFF5A1F),
    NECROMANCER("Некромант", 0x7B2CBF),
    CHRONOMANCER("Хрономант", 0x2EC4F1),
    DRUID("Друид", 0x4CAF50);

    public final String title;
    public final int color;

    MagicPath(String title, int color) {
        this.title = title;
        this.color = color;
    }

    public static MagicPath byId(int id) {
        MagicPath[] v = values();
        return id >= 0 && id < v.length ? v[id] : NONE;
    }
}

// ===== capability/PlayerMana.java =====
package com.aetherium.overlord.capability;

import net.minecraft.nbt.CompoundTag;
import net.minecraftforge.common.util.INBTSerializable;

public class PlayerMana implements INBTSerializable<CompoundTag> {
    private float mana = 100f;
    private float baseMax = 100f;
    private float regenPerSecond = 2f;
    private MagicPath path = MagicPath.NONE;
    private int pathLevel = 0;
    private int pathPoints = 0;
    private float lastSynced = -1f;
    // не сохраняется: считается каждый тик из брони/артефактов (1.0 = без бонуса, 6.0 = +500%)
    private float maxMultiplier = 1f;

    public float getMana() { return mana; }
    public float getEffectiveMax() { return baseMax * maxMultiplier; }
    public float getBaseMax() { return baseMax; }
    public float getRegen() { return regenPerSecond * (1f + (maxMultiplier - 1f) * 0.2f); }
    public MagicPath getPath() { return path; }
    public int getPathLevel() { return pathLevel; }
    public int getPathPoints() { return pathPoints; }
    public boolean isPathChosen() { return path != MagicPath.NONE; }

    public void setMaxMultiplier(float m) { this.maxMultiplier = Math.max(1f, m); }

    public void setPath(MagicPath p) {
        if (this.path == MagicPath.NONE) {
            this.path = p;
            this.pathLevel = 1;
            this.baseMax += 25f;
        }
    }

    public boolean consume(float amount) {
        if (mana < amount) return false;
        mana -= amount;
        return true;
    }

    public void addMana(float amount) { mana = Math.min(getEffectiveMax(), mana + amount); }

    public void addPathPoints(int pts) { pathPoints += pts; }

    /** Прокачка: тратит очки, +25 к лимиту маны и +0.25 к регену за уровень. */
    public boolean levelUp() {
        int cost = pathLevel + 1;
        if (pathLevel >= 10 || pathPoints < cost) return false;
        pathPoints -= cost;
        pathLevel++;
        baseMax += 25f;
        regenPerSecond += 0.25f;
        return true;
    }

    public void tick() {
        float max = getEffectiveMax();
        if (mana < max) mana = Math.min(max, mana + getRegen() / 20f);
        if (mana > max) mana = max;
    }

    /** true, если значение изменилось и пора слать пакет клиенту */
    public boolean needsSync() {
        if (Math.abs(mana - lastSynced) >= 0.5f) { lastSynced = mana; return true; }
        return false;
    }
    public void forceSync() { lastSynced = -1f; }

    public void copyFrom(PlayerMana o) {
        this.mana = o.mana; this.baseMax = o.baseMax; this.regenPerSecond = o.regenPerSecond;
        this.path = o.path; this.pathLevel = o.pathLevel; this.pathPoints = o.pathPoints;
    }

    @Override
    public CompoundTag serializeNBT() {
        CompoundTag t = new CompoundTag();
        t.putFloat("mana", mana);
        t.putFloat("baseMax", baseMax);
        t.putFloat("regen", regenPerSecond);
        t.putInt("path", path.ordinal());
        t.putInt("level", pathLevel);
        t.putInt("points", pathPoints);
        return t;
    }

    @Override
    public void deserializeNBT(CompoundTag t) {
        mana = t.getFloat("mana");
        baseMax = t.contains("baseMax") ? t.getFloat("baseMax") : 100f;
        regenPerSecond = t.contains("regen") ? t.getFloat("regen") : 2f;
        path = MagicPath.byId(t.getInt("path"));
        pathLevel = t.getInt("level");
        pathPoints = t.getInt("points");
    }
}

// ===== capability/PlayerManaProvider.java =====
package com.aetherium.overlord.capability;

import net.minecraft.core.Direction;
import net.minecraft.nbt.CompoundTag;
import net.minecraftforge.common.capabilities.Capability;
import net.minecraftforge.common.capabilities.ICapabilitySerializable;
import net.minecraftforge.common.util.LazyOptional;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.Nullable;

public class PlayerManaProvider implements ICapabilitySerializable<CompoundTag> {
    private final PlayerMana mana = new PlayerMana();
    private final LazyOptional<PlayerMana> optional = LazyOptional.of(() -> mana);

    @Override
    public <T> @NotNull LazyOptional<T> getCapability(@NotNull Capability<T> cap, @Nullable Direction side) {
        return cap == ModCapabilities.PLAYER_MANA ? optional.cast() : LazyOptional.empty();
    }

    @Override public CompoundTag serializeNBT() { return mana.serializeNBT(); }
    @Override public void deserializeNBT(CompoundTag nbt) { mana.deserializeNBT(nbt); }
}

package com.aetherium.overlord.network;

import com.aetherium.overlord.AetheriumOverlordMod;
import com.aetherium.overlord.capability.ModCapabilities;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.server.level.ServerPlayer;
import net.minecraftforge.network.NetworkRegistry;
import net.minecraftforge.network.PacketDistributor;
import net.minecraftforge.network.simple.SimpleChannel;

public class ModNetwork {
    private static final String PROTOCOL = "1";
    public static final SimpleChannel CHANNEL = NetworkRegistry.newSimpleChannel(
            new ResourceLocation(AetheriumOverlordMod.MODID, "main"),
            () -> PROTOCOL, PROTOCOL::equals, PROTOCOL::equals);

    public static void register() {
        int id = 0;
        CHANNEL.registerMessage(id++, ManaSyncPacket.class,
                ManaSyncPacket::encode, ManaSyncPacket::decode, ManaSyncPacket::handle);
        CHANNEL.registerMessage(id++, ChoosePathPacket.class,
                ChoosePathPacket::encode, ChoosePathPacket::decode, ChoosePathPacket::handle);
    }

    public static void syncMana(ServerPlayer player) {
        player.getCapability(ModCapabilities.PLAYER_MANA).ifPresent(m ->
                CHANNEL.send(PacketDistributor.PLAYER.with(() -> player),
                        new ManaSyncPacket(m.getMana(), m.getEffectiveMax(),
                                m.getPath().ordinal(), m.getPathLevel(), m.getPathPoints())));
    }
}

// ===== network/ManaSyncPacket.java =====
package com.aetherium.overlord.network;

import com.aetherium.overlord.client.ClientManaData;
import net.minecraft.network.FriendlyByteBuf;
import net.minecraftforge.api.distmarker.Dist;
import net.minecraftforge.fml.DistExecutor;
import net.minecraftforge.network.NetworkEvent;
import java.util.function.Supplier;

public record ManaSyncPacket(float mana, float max, int path, int level, int points) {
    public static void encode(ManaSyncPacket p, FriendlyByteBuf buf) {
        buf.writeFloat(p.mana); buf.writeFloat(p.max);
        buf.writeInt(p.path); buf.writeInt(p.level); buf.writeInt(p.points);
    }
    public static ManaSyncPacket decode(FriendlyByteBuf buf) {
        return new ManaSyncPacket(buf.readFloat(), buf.readFloat(), buf.readInt(), buf.readInt(), buf.readInt());
    }
    public static void handle(ManaSyncPacket p, Supplier<NetworkEvent.Context> ctx) {
        ctx.get().enqueueWork(() -> DistExecutor.unsafeRunWhenOn(Dist.CLIENT,
                () -> () -> ClientManaData.set(p.mana, p.max, p.path, p.level, p.points)));
        ctx.get().setPacketHandled(true);
    }
}

// ===== network/ChoosePathPacket.java =====
package com.aetherium.overlord.network;

import com.aetherium.overlord.capability.MagicPath;
import com.aetherium.overlord.capability.ModCapabilities;
import com.aetherium.overlord.item.PathTomeItem;
import net.minecraft.network.FriendlyByteBuf;
import net.minecraft.server.level.ServerPlayer;
import net.minecraftforge.network.NetworkEvent;
import java.util.function.Supplier;

public record ChoosePathPacket(int path) {
    public static void encode(ChoosePathPacket p, FriendlyByteBuf buf) { buf.writeInt(p.path); }
    public static ChoosePathPacket decode(FriendlyByteBuf buf) { return new ChoosePathPacket(buf.readInt()); }

    public static void handle(ChoosePathPacket p, Supplier<NetworkEvent.Context> ctx) {
        ctx.get().enqueueWork(() -> {
            ServerPlayer player = ctx.get().getSender();
            if (player == null || !(player.getMainHandItem().getItem() instanceof PathTomeItem)) return;
            MagicPath chosen = MagicPath.byId(p.path);
            if (chosen == MagicPath.NONE) return;
            player.getCapability(ModCapabilities.PLAYER_MANA).ifPresent(m -> {
                if (m.isPathChosen()) return;
                m.setPath(chosen);
                m.addPathPoints(1);
                m.forceSync();
                player.getMainHandItem().shrink(1);
                ModNetwork.syncMana(player);
            });
        });
        ctx.get().setPacketHandled(true);
    }
}

package com.aetherium.overlord.events;

import com.aetherium.overlord.AetheriumOverlordMod;
import com.aetherium.overlord.capability.*;
import com.aetherium.overlord.network.ModNetwork;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.entity.player.Player;
import net.minecraftforge.event.AttachCapabilitiesEvent;
import net.minecraftforge.event.TickEvent;
import net.minecraftforge.event.entity.player.PlayerEvent;
import net.minecraftforge.eventbus.api.SubscribeEvent;
import net.minecraftforge.fml.common.Mod;

@Mod.EventBusSubscriber(modid = AetheriumOverlordMod.MODID)
public class ManaEvents {

    @SubscribeEvent
    public static void attach(AttachCapabilitiesEvent<Entity> e) {
        if (e.getObject() instanceof Player) {
            e.addCapability(new ResourceLocation(AetheriumOverlordMod.MODID, "mana"), new PlayerManaProvider());
        }
    }

    @SubscribeEvent
    public static void clone(PlayerEvent.Clone e) {
        e.getOriginal().reviveCaps();
        e.getOriginal().getCapability(ModCapabilities.PLAYER_MANA).ifPresent(old ->
                e.getEntity().getCapability(ModCapabilities.PLAYER_MANA).ifPresent(n -> n.copyFrom(old)));
        e.getOriginal().invalidateCaps();
    }

    @SubscribeEvent
    public static void tick(TickEvent.PlayerTickEvent e) {
        if (e.phase != TickEvent.Phase.END || e.player.level().isClientSide) return;
        e.player.getCapability(ModCapabilities.PLAYER_MANA).ifPresent(m -> {
            m.setMaxMultiplier(ArmorBonuses.manaMultiplier(e.player)); // см. раздел 3
            m.tick();
            if (e.player.tickCount % 10 == 0 && m.needsSync()) ModNetwork.syncMana((ServerPlayer) e.player);
        });
    }

    @SubscribeEvent
    public static void login(PlayerEvent.PlayerLoggedInEvent e) {
        if (e.getEntity() instanceof ServerPlayer sp) { sp.getCapability(ModCapabilities.PLAYER_MANA).ifPresent(PlayerMana::forceSync); ModNetwork.syncMana(sp); }
    }
    @SubscribeEvent
    public static void respawn(PlayerEvent.PlayerRespawnEvent e) {
        if (e.getEntity() instanceof ServerPlayer sp) ModNetwork.syncMana(sp);
    }
    @SubscribeEvent
    public static void dimension(PlayerEvent.PlayerChangedDimensionEvent e) {
        if (e.getEntity() instanceof ServerPlayer sp) ModNetwork.syncMana(sp);
    }
}

public class ArmorBonuses {
    public static float manaMultiplier(Player p) {
        boolean full = p.getItemBySlot(EquipmentSlot.HEAD).is(ModItems.ARCHMAGE_HELMET.get())
                && p.getItemBySlot(EquipmentSlot.CHEST).is(ModItems.ARCHMAGE_CHESTPLATE.get())
                && p.getItemBySlot(EquipmentSlot.LEGS).is(ModItems.ARCHMAGE_LEGGINGS.get())
                && p.getItemBySlot(EquipmentSlot.FEET).is(ModItems.ARCHMAGE_BOOTS.get());
        return full ? 6f : 1f; // 6.0 = +500%
    }
}

package com.aetherium.overlord.client;

import com.aetherium.overlord.AetheriumOverlordMod;
import com.aetherium.overlord.capability.MagicPath;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraftforge.api.distmarker.Dist;
import net.minecraftforge.client.event.RegisterGuiOverlaysEvent;
import net.minecraftforge.client.gui.overlay.VanillaGuiOverlay;
import net.minecraftforge.eventbus.api.SubscribeEvent;
import net.minecraftforge.fml.common.Mod;

@Mod.EventBusSubscriber(modid = AetheriumOverlordMod.MODID, bus = Mod.EventBusSubscriber.Bus.MOD, value = Dist.CLIENT)
public class ManaHud {

    @SubscribeEvent
    public static void register(RegisterGuiOverlaysEvent e) {
        e.registerAbove(VanillaGuiOverlay.PLAYER_HEALTH.id(), "mana_bar", (gui, g, partialTick, w, h) -> {
            Minecraft mc = Minecraft.getInstance();
            if (mc.player == null || mc.options.hideGui || mc.player.isSpectator()) return;
            draw(g, w, h, mc);
        });
    }

    private static void draw(GuiGraphics g, int w, int h, Minecraft mc) {
        int barW = 81, barH = 7;
        int x = w / 2 - 91, y = h - 59;
        float ratio = ClientManaData.max <= 0 ? 0 : Math.min(1f, ClientManaData.mana / ClientManaData.max);
        int color = 0xFF000000 | MagicPath.byId(ClientManaData.path).color;
        if (ClientManaData.path == 0) color = 0xFF3A7BFF;

        g.fill(x - 1, y - 1, x + barW + 1, y + barH + 1, 0xFF101018);   // рамка
        g.fill(x, y, x + barW, y + barH, 0xFF1C1C2A);                   // фон
        int filled = (int) (barW * ratio);
        g.fill(x, y, x + filled, y + barH, color);                       // заполнение
        g.fill(x, y, x + filled, y + 2, 0x66FFFFFF);                     // блик
        String text = (int) ClientManaData.mana + " / " + (int) ClientManaData.max;
        g.drawString(mc.font, text, x + barW + 4, y, 0xFFE8E8FF, true);
        if (ClientManaData.path != 0) {
            g.drawString(mc.font, MagicPath.byId(ClientManaData.path).title + " ур." + ClientManaData.level
                    + "  [очки: " + ClientManaData.points + "]", x, y - 10, color, true);
        }
    }
}

        super.render(g, mx, my, pt);
    }

    @Override public boolean isPauseScreen() { return false; }

--- 1.2 Механика прокачки пути ---
 • Уровни пути: 1–10. Стоимость уровня N = N очков пути (ур.1→2 стоит 2, ... 9→10 стоит 10).
 • Очки пути: награды квестов, Осколки Пути (предмет), убийство боссов (+3..+10).
 • Каждый уровень: +25 к макс. мане, +0.25 маны/сек. Сет Архимага: +500% лимита маны.
 • Выбор пути — один раз, через Фолиант Выбора Пути (крафт: книга + 4 эссенции, либо стартовый предмет через FTB Quests).
 • Посох чужого пути работает только в творческом режиме либо с Руной Универсала (крафт в Столе начертания).

--- 1.3 Четыре Пути ---

=== ПУТЬ 1: ПИРОМАНТ / РАЗРУШИТЕЛЬ (цвет: оранжевый) ===
Стиль: максимальный урон по площади, риск и награда.
Древо прокачки:
 Ур.1  Искра — +10% урона огнём, базовые посохи Пути.
 Ур.2  Тепло Очага — враги рядом (3 бл.) получают поджог.
 Ур.3  Огнеупорность — полный иммунитет к огню и лаве.
 Ур.4  Магмовый След — за вами остаётся горящий след (урон 2/с).
 Ур.5  Щит Пепла — поглощает до 8 урона раз в 30 с.
 Ур.6  Жар Души — огненные заклинания +30% радиус.
 Ур.7  Метеорная Метка — каждое 5-е заклинание вызывает малый метеор.
 Ур.8  Феникс — при смерти воскрешает (КД 20 минут), взрыв радиус 6.
 Ур.9  Сердце Вулкана — регенерация маны x2 в Незере и около огня.
 Ур.10 УЛЬТИМЕЙТ «Метеоритный Апокалипсис»: 12 метеоритов в радиусе 20 бл. на 15 с (урон 40 каждый). Затраты 400 маны, КД 3 минуты.
Пассивки (всегда): +20% ко всему огненному урону; враги в огне получают -15% брони.

=== ПУТЬ 2: НЕКРОМАНТ / ПОВЕЛИТЕЛЬ БЕЗДНЫ (цвет: фиолетовый) ===
Стиль: армия миньонов, выживаемость, контроль.
 Ур.1  Шёпот Могил — вампиризм 10% от магического урона.
 Ур.2  Слуга Праха — лимит миньонов 2.
 Ур.3  Проклятие Слабости — цель наносит -30% урона 10 с.
 Ур.4  Тёмные Узы — миньоны получают вашу броню x0.5.
 Ур.5  Пожиратель — убийство даёт +4 маны и +2 HP.
 Ур.6  Чумная Аура — враги рядом получают отравление, миньоны не страдают.
 Ур.7  Командир Мёртвых — лимит миньонов 6, миньоны сильнее на 50%.
 Ур.8  Проклятие Увядания — цель теряет 8% макс. HP/с 5 с.
 Ур.9  Бессмертный Лич — смертельный урон вместо смерти: 3 с неуязвимости, -50% маны (КД 15 мин).

package com.aetherium.overlord.item;

import com.aetherium.overlord.capability.MagicPath;
import com.aetherium.overlord.capability.ModCapabilities;
import com.aetherium.overlord.capability.PlayerMana;

import net.minecraft.world.InteractionHand;
import net.minecraft.world.InteractionResultHolder;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.Level;
import net.minecraft.world.phys.Vec3;

import java.util.Optional;

public abstract class MagicStaffItem extends Item {
    protected final MagicPath path;
    protected final int tier;
    protected final float manaCost, ultimateCost;
    protected final int cooldown, ultimateCooldown;   // в тиках

    public MagicStaffItem(Properties props, MagicPath path, int tier,
                          float manaCost, int cooldownTicks, float ultimateCost, int ultimateCooldownTicks) {
        super(props); // в Properties задавайте .durability(N) (это уже даёт стак 1)
        this.path = path; this.tier = tier;
        this.manaCost = manaCost; this.cooldown = cooldownTicks;
        this.ultimateCost = ultimateCost; this.ultimateCooldown = ultimateCooldownTicks;
    }

    /** ПКМ. Вернуть true, если заклинание сработало. */
    protected abstract boolean castPrimary(ServerLevel level, ServerPlayer player, ItemStack stack);
    /** Shift+ПКМ. */
    protected abstract boolean castUltimate(ServerLevel level, ServerPlayer player, ItemStack stack);

    @Override
    public InteractionResultHolder<ItemStack> use(Level level, Player player, InteractionHand hand) {
        ItemStack stack = player.getItemInHand(hand);
        if (level.isClientSide) return InteractionResultHolder.sidedSuccess(stack, true);
        if (player.getCooldowns().isOnCooldown(this)) return InteractionResultHolder.fail(stack);

        ServerPlayer sp = (ServerPlayer) player;
        ServerLevel sl = (ServerLevel) level;
        Optional<PlayerMana> opt = sp.getCapability(ModCapabilities.PLAYER_MANA).resolve();
        if (opt.isEmpty()) return InteractionResultHolder.fail(stack);
        PlayerMana mana = opt.get();

        boolean creative = sp.isCreative();
        if (!creative && path != MagicPath.NONE && mana.getPath() != path) {
            sp.displayClientMessage(Component.literal("§cЭтот посох принадлежит Пути: " + path.title), true);
            return InteractionResultHolder.fail(stack);
        }

        boolean ult = sp.isShiftKeyDown();
        float cost = ult ? ultimateCost : manaCost;
        if (!creative && !mana.consume(cost)) {
            sp.displayClientMessage(Component.literal("§9Не хватает маны (" + (int) cost + ")"), true);
            return InteractionResultHolder.fail(stack);
        }

        boolean ok = ult ? castUltimate(sl, sp, stack) : castPrimary(sl, sp, stack);
        if (!ok) {
            if (!creative) mana.addMana(cost); // возврат
            return InteractionResultHolder.fail(stack);
        }
        stack.hurtAndBreak(ult ? 3 : 1, sp, p -> p.broadcastBreakEvent(hand));
        sp.getCooldowns().addCooldown(this, ult ? ultimateCooldown : cooldown);
        ModNetwork.syncMana(sp);
        return InteractionResultHolder.sidedSuccess(stack, false);
    }

    /** Выстрел кастомным снарядом из глаз игрока. */
    protected void shootBolt(ServerLevel level, ServerPlayer player, float damage, float aoeRadius,
                             ParticleOptions particle, float speed) {
        Vec3 look = player.getLookAngle();
        MagicBoltProjectile bolt = new MagicBoltProjectile(level, player, look.x, look.y, look.z);
        bolt.setPos(player.getX() + look.x, player.getEyeY() - 0.1 + look.y, player.getZ() + look.z);
        bolt.configure(damage, aoeRadius, particle, path);
        bolt.shoot(look.x, look.y, look.z, speed, 0f);
        level.addFreshEntity(bolt);
    }

    protected float scaledDamage(float base) { return base * (1f + tier * 0.35f); }

    public MagicPath getPath() { return path; }
    public int getTier() { return tier; }
}

package com.aetherium.overlord.entity;

import com.aetherium.overlord.capability.MagicPath;
import com.aetherium.overlord.registry.ModEntities;
import net.minecraft.core.particles.ParticleOptions;
import net.minecraft.core.particles.ParticleTypes;
import net.minecraft.server.level.ServerLevel;

import net.minecraft.world.entity.EntityType;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.projectile.AbstractHurtingProjectile;
import net.minecraft.world.level.Level;
import net.minecraft.world.phys.EntityHitResult;
import net.minecraft.world.phys.HitResult;
import net.minecraft.world.phys.AABB;

public class MagicBoltProjectile extends AbstractHurtingProjectile {
    private float damage = 6f, aoe = 0f;
    private ParticleOptions particle = ParticleTypes.ENCHANT;
    private MagicPath element = MagicPath.NONE;
    private int life = 0;

    public MagicBoltProjectile(EntityType<? extends MagicBoltProjectile> type, Level level) { super(type, level); }

    public MagicBoltProjectile(Level level, LivingEntity owner, double dx, double dy, double dz) {
        super(ModEntities.MAGIC_BOLT.get(), owner, dx, dy, dz, level);
    }

    public void configure(float damage, float aoe, ParticleOptions particle, MagicPath element) {
        this.damage = damage; this.aoe = aoe; this.particle = particle; this.element = element;
    }

    @Override protected ParticleOptions getTrailParticle() { return particle; }
    @Override protected boolean shouldBurn() { return false; }
    @Override public boolean isOnFire() { return false; }

    @Override
    public void tick() {
        super.tick();
        if (!level().isClientSide && ++life > 80) discard();
        if (level().isClientSide) {
            level().addParticle(particle, getX(), getY(), getZ(), 0, 0, 0);
        }
    }

    @Override
    protected void onHitEntity(EntityHitResult res) {
        super.onHitEntity(res);
        if (level().isClientSide) return;
        res.getEntity().hurt(damageSources().indirectMagic(this, getOwner()), damage);
        if (element == MagicPath.PYROMANCER) res.getEntity().setSecondsOnFire(4);
    }

    @Override
    protected void onHit(HitResult res) {
        super.onHit(res);
        if (level().isClientSide) return;
        if (aoe > 0f) {
            AABB box = getBoundingBox().inflate(aoe);
            for (LivingEntity e : level().getEntitiesOfClass(LivingEntity.class, box, e -> e != getOwner())) {
                e.hurt(damageSources().indirectMagic(this, getOwner()), damage * 0.6f);
            }
            ((ServerLevel) level()).sendParticles(particle, getX(), getY(), getZ(), 40, aoe / 2, aoe / 2, aoe / 2, 0.05);
        }
        discard();
    }
}

package com.aetherium.overlord.item;

import com.aetherium.overlord.capability.MagicPath;
import net.minecraft.core.particles.ParticleOptions;
import net.minecraft.core.particles.ParticleTypes;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.effect.MobEffectInstance;
import net.minecraft.world.effect.MobEffects;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.Mob;
import net.minecraft.world.entity.monster.Vex;
import net.minecraft.world.entity.EntityType;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.phys.AABB;
import net.minecraft.world.phys.Vec3;

public class ElementalStaff extends MagicStaffItem {
    private final float baseDamage;

    public ElementalStaff(Properties p, MagicPath path, int tier, float mana, int cd, float ultMana, int ultCd, float baseDamage) {
        super(p, path, tier, mana, cd, ultMana, ultCd);
        this.baseDamage = baseDamage;
    }

    private ParticleOptions particle() {
        return switch (path) {
            case PYROMANCER -> ParticleTypes.FLAME;
            case NECROMANCER -> ParticleTypes.SOUL_FIRE_FLAME;
            case CHRONOMANCER -> ParticleTypes.END_ROD;
            case DRUID -> ParticleTypes.HAPPY_VILLAGER;
            default -> ParticleTypes.ENCHANT;
        };
    }

    @Override
    protected boolean castPrimary(ServerLevel level, ServerPlayer player, ItemStack stack) {
        float aoe = path == MagicPath.PYROMANCER ? 1.5f + tier * 0.4f : 0f;
        shootBolt(level, player, scaledDamage(baseDamage), aoe, particle(), 1.6f);
        if (path == MagicPath.NECROMANCER) player.heal(scaledDamage(baseDamage) * 0.15f); // вампиризм
        if (path == MagicPath.DRUID) player.heal(1f + tier);
        return true;
    }

    @Override
    protected boolean castUltimate(ServerLevel level, ServerPlayer player, ItemStack stack) {
        double r = 6 + tier * 2;
        AABB box = player.getBoundingBox().inflate(r);
        switch (path) {
            case PYROMANCER -> {
                // метеоритный дождь: снаряды падают сверху вокруг игрока
                for (int i = 0; i < 4 + tier * 2; i++) {
                    double x = player.getX() + (level.random.nextDouble() - 0.5) * r * 2;
                    double z = player.getZ() + (level.random.nextDouble() - 0.5) * r * 2;
                    var m = new com.aetherium.overlord.entity.MagicBoltProjectile(level, player, 0, -1, 0);
                    m.setPos(x, player.getY() + 18, z);
                    m.configure(scaledDamage(baseDamage) * 2f, 3f, ParticleTypes.LAVA, path);
                    m.shoot(0, -1, 0, 1.2f, 0f);
                    level.addFreshEntity(m);
                }
            }
            case NECROMANCER -> {
                for (int i = 0; i < 2 + tier; i++) {
                    Vex v = EntityType.VEX.create(level);
                    if (v == null) continue;
                    v.moveTo(player.getX() + i - 1, player.getY() + 1, player.getZ());
                    v.setOwner(player);
                    v.setLimitedLife(20 * (40 + tier * 10));
                    level.addFreshEntity(v);
                }
            }
            case CHRONOMANCER -> {
                for (LivingEntity e : level.getEntitiesOfClass(LivingEntity.class, box, e -> e != player)) {
                    e.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 20 * 5, 4));
                    e.addEffect(new MobEffectInstance(MobEffects.WEAKNESS, 20 * 5, 1));
                }
                Vec3 look = player.getLookAngle().scale(8);
                player.teleportTo(player.getX() + look.x, player.getY(), player.getZ() + look.z); // блинк
            }
            case DRUID -> {
                for (LivingEntity e : level.getEntitiesOfClass(LivingEntity.class, box)) {
                    if (e == player || e instanceof net.minecraft.world.entity.player.Player) {
                        e.heal(6f + tier * 2);
                        e.addEffect(new MobEffectInstance(MobEffects.REGENERATION, 20 * 8, 1));
                    } else if (e instanceof Mob) {
                        e.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 20 * 4, 6)); // корни
                    }
                }
            }
            default -> {
                for (LivingEntity e : level.getEntitiesOfClass(LivingEntity.class, box, e -> e != player)) {
                    e.hurt(player.damageSources().indirectMagic(player, player), scaledDamage(baseDamage));
                }
            }
        }
        level.sendParticles(particle(), player.getX(), player.getY() + 1, player.getZ(), 80, r / 3, 1, r / 3, 0.05);
        return true;
    }
}

--- 5.1 ТЕХНИЧЕСКОЕ ЗАДАНИЕ (ТЗ) ДЛЯ ХУДОЖНИКА ---
Общие требования:
 • Программа: Blockbench 4.x, формат проекта "GeckoLib Model" (animated model) для предметов/брони/мобов.
 • Единицы: 1 блок = 16 px. Текстуры PNG, степень двойки: посохи 64x64 (3D) или 128x128 (крупные), броня 128x128, боссы 256x256.
 • Имена костей (bones) только латиницей, snake_case: root, crystal, crystal_ring_1, head, horns, wing_left, wing_right...
 • Один-единственный root в корне модели. Центр pivot — в точке вращения кости.
 • Лимит: посохи ≤ 150 кубов, броня ≤ 400 кубов на комплект, босс ≤ 1200 кубов.
 • Файлы: geo — assets/<modid>/geo/..., анимации — assets/<modid>/animations/..., текстуры — assets/<modid>/textures/...
 • Эмиссив (светящиеся части): отдельная текстура *_glowing.png, подключается через GeoRenderLayer (AutoGlowingGeoLayer).

Посохи (ключевые модели):
 А. Посох Звёздопада Разрушителя — длина 2.5 блока, оранжевое ядро-лава в навершии, парящие осколки (3 шт.), анимация: вращение ядра (360° за 4 с), пульсация света.
 Б. Посох Бездны — чёрный скелетоподобный посох с черепом, внутри глаза светятся фиолетовым; анимация: моргание глаз, дрожь.
 В. Посох Хроноса — песочные часы-навершие, песок "течёт"; анимация: перелив песка, вращение стрелок.
 Г. Посох Древа Жизни — живая ветвь, цветущая верхушка; анимация: покачивание, раскрытие цветка при касте.
 Д. Посох Создателя Миров — золотой с радужными кольцами; три вращающихся кольца в разных осях; кристалл-ядро с анимацией дыхания.
 Анимации для каждого: idle (loop), cast (0.5 с), ultimate (1.5 с, вспышка).

Броня (3D):
 • Нагрудник Хроно/Покрова Пустоты — крылья (wing_left/wing_right) по 4 сегмента перьев; анимации: idle (лёгкое колыхание), fly (взмах 1 с), land.
 • Шлем Савана Бездны — изогнутые рога (horns) + светящиеся глаза в капюшоне.
 • Шлем Эфирного Архимага — парящая над головой корона-кольцо с вращением.
 • Требование: расчёт размеров под стандартный скин Steve/Alex; анимации через GeoArmorRenderer.

Боссы (кастомные мобы):
 1. СТРАЖ АКАДЕМИИ ЭФИРА — каменный голем 4 блока высотой, 3 руки-кристалла. Анимации: idle, walk, slam (удар в землю), beam (луч), spawn, death.
 2. ХРАНИТЕЛЬ ЭФИРА — парящая фигура с рогами и 4 крылья. Фазы: огонь, бездна, время, природа — меняет цвет. Анимации: idle_float, attack_melee, attack_ranged, phase_change, death.
 3. ДРЕВНИЙ БОГ ЭФИРА — финальный босс, 12 блоков высотой, множество глаз/щупалец. Анимации: idle, tentacle_sweep, eye_beam, summon, enrage, death.
 4. Мини-босс: ЛИЧ КЛЕПСИДРЫ — для склепа некромантов.

// ===== item/CrystalStaffItem.java =====
package com.aetherium.overlord.item;

import com.aetherium.overlord.capability.MagicPath;
import com.aetherium.overlord.client.renderer.CrystalStaffRenderer;
import net.minecraft.client.renderer.BlockEntityWithoutLevelRenderer;
import net.minecraftforge.client.extensions.common.IClientItemExtensions;
import software.bernie.geckolib.animatable.GeoItem;
import software.bernie.geckolib.animatable.SingletonGeoAnimatable;
import software.bernie.geckolib.core.animatable.instance.AnimatableInstanceCache;
import software.bernie.geckolib.core.animation.AnimatableManager;
import software.bernie.geckolib.core.animation.AnimationController;
import software.bernie.geckolib.core.animation.RawAnimation;
import software.bernie.geckolib.core.object.PlayState;
import software.bernie.geckolib.util.GeckoLibUtil;

import java.util.function.Consumer;

public class CrystalStaffItem extends ElementalStaff implements GeoItem {
    private static final RawAnimation SPIN = RawAnimation.begin().thenLoop("animation.staff.spin");
    private final AnimatableInstanceCache cache = GeckoLibUtil.createInstanceCache(this);

    public CrystalStaffItem(Properties p, MagicPath path, int tier, float mana, int cd, float ultMana, int ultCd, float dmg) {
        super(p, path, tier, mana, cd, ultMana, ultCd, dmg);
        SingletonGeoAnimatable.registerSyncedAnimatable(this);
    }

    @Override
    public void initializeClient(Consumer<IClientItemExtensions> consumer) {
        consumer.accept(new IClientItemExtensions() {
            private CrystalStaffRenderer renderer;
            @Override
            public BlockEntityWithoutLevelRenderer getCustomRenderer() {
                if (this.renderer == null) this.renderer = new CrystalStaffRenderer();
                return this.renderer;
            }
        });
    }

    @Override
    public void registerControllers(AnimatableManager.ControllerRegistrar controllers) {
        controllers.add(new AnimationController<>(this, "spin_controller", 0,
                state -> state.setAndContinue(SPIN)));
    }

    @Override
    public AnimatableInstanceCache getAnimatableInstanceCache() { return this.cache; }
}

РАЗДЕЛ 7. ГЕНЕРАЦИЯ МИРА И СТРУКТУРЫ (12 типов)
================================================================

--- 7.1 Биомы ---
 1. Эфирные Рощи (Верхний мир) — светящаяся трава, зачарованные деревья, мана-мох, лунные лилии. Мобы: эфирные олени (нейтр.), светлячки-духи. Редкий биом, рядом с ним чаще появляются руды эфира.
 2. Кристальная Пустошь (Верхний мир, горы) — кластеры кристаллов 4 цветов, выходы руды Самоцвета Маны. Мобы: кристальные големы (враждебные), каменные жуки.
 3. Выжженный Разлом (Незер) — трещины с эфирным светом, руда Тлеющего Угля. Мобы: пепельные элементали, обезображенные призраки.
 4. Эфирное Измерение — парящие острова, море света, Арена Бога в центре.

--- 7.2 Структуры ---
 1. ЛЕТАЮЩАЯ АКАДЕМИЯ МАГОВ
    Где: небо над Верхним миром, высота Y=150–220. Редкость: 1 на ~2000x2000 блоков.
    Размер: главный остров 60x60, башни до 40 блоков, 3–5 малых островов.
    Лут: сундуки библиотеки — свитки, пустые руны, самоцветы маны, гримуары T1–T2; в хранилище — посох T2–T3.
    Мобы: призрачные ученики (стреляют искрой), магические стражи, мини-босс Страж Академии (Арена в центре, Квест №12).
 2. ПОДЗЕМНЫЙ СКЛЕП НЕКРОМАНТА
    Где: под землёй Y=-50…10. Редкость: 1 на ~1500 блоков.
    Размер: 40x40, 3 уровня, катакомбы.
    Лут: осколки души, кости/мясо гнили, Гримуар Костяной Жатвы, посох некро T1–T3, редкий амулет Души.
    Мобы: скелеты-маги, зомби-слуги, призрачные векса, мини-босс Лич Клепсидры.
 3. ОБЕЗОБРАЖЕННЫЙ РАЗЛОМ
    Где: Верхний мир (поверхность, как трещина 80 блоков) и в Незере. Редкость: 1 на ~2500 блоков.
    Особенности: гравитационные аномалии, обезображенные блоки (случайная замена текстуры), портал вниз в мини-измерение.
    Лут: эндерит, кровавая ржавь, рунные обломки, артефакты T2.
    Мобы: обезображенные (искажённые версии зомби/скелетов), тени, Страж Разлома.
 4. АРЕНА СТРАЖА ЭФИРА (Босс 1)
    Где: центр Летающей Академии, закрытая арена 30x30 (активируется Алтарём призыва).
    Бой: 3 фазы — удар кулаками, лучи кристаллов, призыв каменных големов.
    Лут: посох T3, Руна Универсала, 3 очка пути, шанс Гримуара Архимага.
 5. БАШНЯ ХРОНОМАНТА
    Где: равнины/пустыня; башня 12x12 высотой 50, внутри — часовой механизм.
    Лут: хронокристаллы, песочные посохи, кольцо Песка Времени.
    Мобы: часовые автоматоны (замедляют), хрономаг-призрак.
 6. РОЩА ДРУИДОВ (Древо Жизни)
    Где: биом Эфирные Рощи/тёмный лес. Гигантское древо (40 блоков) и круг камней.
    Лут: эссенция природы, посохи друида T1–T3, плащ Хранителя.
    Мобы: энты-стражи (нейтральные, агрессивны при порче рощи), дриады.
 7. ВУЛКАНИЧЕСКИЙ АЛТАРЬ ПИРОМАНТОВ
    Где: Незер, около лавовых озёр. Платформа с лавовым фонтаном.
    Лут: эссенция огня, посох пиро T2–T4, сеты Пепельного Пиромантa (части).
    Мобы: огненные элементали, магма-големы, Огненный Страж (мини-босс).
 8. ЭНДСКАЯ ЦИТАДЕЛЬ ЭФИРА
    Где: внешние острова Энда. Фиолетово-золотая башня 70 блоков.
    Лут: эндерит, эфирный сплав, части Покрова Пустоты, реликвии.
    Мобы: стражи пустоты, эндер-маги, босс Хранитель Эфира (Босс 2, Квест №21).
 9. ЗАТОПЛЕННАЯ БИБЛИОТЕКА
    Где: океаны, дно. Стеклянный купол, магическое стекло.
    Лут: свитки телепорта, пустые руны, гримуары.
    Мобы: утопшие маги, стражи вод.
 10. РУННЫЙ ОБЕЛИСК (малая структура)
    Где: любой биом, высота 8–15 блоков, много штук.
    Лут: самоцветы маны, руны. Особенность: подзаряжает ману рядом (аура).
 11. ЛАГЕРЬ ОТСТУПНИКОВ
    Где: поверхность, 20x20. Лут: гримуары, зелья.
    Мобы: отступники-маги (человеко-образные), атакуют с посохами.
 12. АРЕНА БОГА (Босс 3, финал)
    Где: центр Эфирного Измерения. Круглая арена 60 блоков с руническими кругами.
    Бой: Древний Бог Эфира — 4 фазы (по стихиям), 1200 HP, призыв, лучи, ярость.
    Лут: Реликвия Древнего Бога, 10 очков пути, 1000 XP, сплав Повелителя.

--- 7.3 Техническая реализация ---
Структуры в 1.20.1 — только данные (датапак): шаблоны NBT + JSON.
 • data/aetherium_overlord/structures/<имя>.nbt        — файлы из блока структуры (сохраняете в игре)
 • data/aetherium_overlord/worldgen/template_pool/*.json
 • data/aetherium_overlord/worldgen/structure/*.json
 • data/aetherium_overlord/worldgen/structure_set/*.json (spacing, separation, salt)
 • data/aetherium_overlord/forge/biome_modifier/*.json   — добавить руды/структуры в существующие биомы

Пример: руда в Верхнем мире (configured + placed feature + biome modifier):

// data/aetherium_overlord/worldgen/configured_feature/ether_ore.json
{
  "type": "minecraft:ore",
  "config": {
    "discard_chance_on_air_exposure": 0.0,
    "size": 6,
    "targets": [
      { "state": { "Name": "aetherium_overlord:ether_ore" },
        "target": { "predicate_type": "minecraft:tag_match", "tag": "minecraft:stone_ore_replaceables" } },
      { "state": { "Name": "aetherium_overlord:deepslate_ether_ore" },
        "target": { "predicate_type": "minecraft:tag_match", "tag": "minecraft:deepslate_ore_replaceables" } }
    ]
  }
}

// data/aetherium_overlord/worldgen/placed_feature/ether_ore.json
{
  "feature": "aetherium_overlord:ether_ore",
  "placement": [
    { "type": "minecraft:count", "count": 6 },
    { "type": "minecraft:in_square" },
    { "type": "minecraft:height_range", "height": { "type": "minecraft:trapezoid", "min_inclusive": { "absolute": -64 }, "max_inclusive": { "absolute": 32 } } },
    { "type": "minecraft:biome" }
  ]
}

// data/aetherium_overlord/forge/biome_modifier/add_ether_ore.json
{
  "type": "forge:add_features",
  "biomes": "#minecraft:is_overworld",
  "features": "aetherium_overlord:ether_ore",
  "step": "underground_ores"
}

--- 8.1 Многоступенчатый крафт: ПОСОХ СОЗДАТЕЛЯ МИРОВ ---
Крафт делится на 6 этапов. Первые этапы — обычные рецепты, этапы 4–6 — Алтарь слияния
(реализуется как собственный тип рецепта или как набор shaped-рецептов на более простом варианте).

Этап 1. Эфирное Ядро (верстак):
   4 Эфирных слитка + 4 Самоцвета маны + 1 Звезда Незера → Эфирное Ядро.
Этап 2. Рунный каркас (Стол начертания рун):
   Эфирный слиток x4 + Руна Огня + Руна Смерти + Руна Времени + Руна Жизни → Рунный Каркас Эфира.
Этап 3. Сплав Повелителя (плавильня/Алтарь слияния):
   Эндерит x4 + Лунное серебро x4 + Хроно-кристалл x2 + Блок эфира → 2 Сплава Повелителя.
Этап 4. Сердце Посоха (Алтарь слияния + 4 пьедестала):
   Эфирное Ядро + Эссенция Огня + Эссенция Бездны + Эссенция Времени + Эссенция Природы → Сердце Стихий.
   (Эссенции: Огня — Огненный Страж, Бездны — Лич Клепсидры, Времени — Башня Хрономанта, Природы — Роща Друидов.)

package com.aetherium.overlord.registry;

import net.minecraft.sounds.SoundEvent;
import net.minecraft.sounds.SoundEvents;
import net.minecraft.world.item.ArmorItem;
import net.minecraft.world.item.ArmorMaterial;
import net.minecraft.world.item.crafting.Ingredient;

public enum ModArmorMaterial implements ArmorMaterial {
    ARCHMAGE("aetherium_overlord:archmage", 38, new int[]{3, 6, 8, 3}, 30, 3.0f, 0.1f);

    private static final int[] BASE_DURABILITY = {13, 15, 16, 11};
    private final String name;
    private final int durabilityMultiplier;
    private final int[] protection;
    private final int enchantability;
    private final float toughness, knockback;

    ModArmorMaterial(String name, int mult, int[] prot, int ench, float tough, float kb) {
        this.name = name; this.durabilityMultiplier = mult; this.protection = prot;
        this.enchantability = ench; this.toughness = tough; this.knockback = kb;
    }

    @Override public int getDurabilityForType(ArmorItem.Type type) { return BASE_DURABILITY[type.ordinal()] * durabilityMultiplier; }
    @Override public int getDefenseForType(ArmorItem.Type type) { return protection[type.ordinal()]; }
    @Override public int getEnchantmentValue() { return enchantability; }
    @Override public SoundEvent getEquipSound() { return SoundEvents.ARMOR_EQUIP_DIAMOND; }
    @Override public Ingredient getRepairIngredient() { return Ingredient.of(ModItems.ETHER_INGOT.get()); }
    @Override public String getName() { return name; }
    @Override public float getToughness() { return toughness; }
    @Override public float getKnockbackResistance() { return knockback; }
}
// Примечание: порядок ArmorItem.Type в 1.20.1: HELMET, CHESTPLATE, LEGGINGS, BOOTS (массивы выше в этом порядке).
// getName() должен совпадать с путём текстур: assets/aetherium_overlord/textures/models/armor/archmage_layer_1.png и _layer_2.png.
// Так как name содержит "modid:name", Forge подставит namespace автоматически.

package com.aetherium.overlord.item;

import com.aetherium.overlord.capability.ModCapabilities;
import com.aetherium.overlord.network.ModNetwork;
import net.minecraft.network.chat.Component;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.InteractionHand;
import net.minecraft.world.InteractionResultHolder;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.Level;

public class PathShardItem extends Item {
    public PathShardItem(Properties p) { super(p); }

    @Override
    public InteractionResultHolder<ItemStack> use(Level level, Player player, InteractionHand hand) {
        ItemStack stack = player.getItemInHand(hand);
        if (!level.isClientSide && player instanceof ServerPlayer sp) {
            sp.getCapability(ModCapabilities.PLAYER_MANA).ifPresent(m -> {
                if (!m.isPathChosen()) {
                    sp.displayClientMessage(Component.literal("§cСначала выбери Путь (фолиант)"), true);
                    return;
                }
                m.addPathPoints(1);
                m.levelUp();            // автоповышение, если хватает очков
                m.forceSync();
                stack.shrink(1);
                ModNetwork.syncMana(sp);
            });
        }
        return InteractionResultHolder.sidedSuccess(stack, level.isClientSide);
    }
}

--- 9.9 .github/workflows/build.yml ---
name: Build Mod

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v3
        with:
          gradle-version: 8.1.1

      - name: Ensure Gradle wrapper
        run: |
          if [ ! -f gradlew ]; then
            gradle wrapper --gradle-version 8.1.1
          fi
          chmod +x gradlew

      - name: Build with Gradle
        run: ./gradlew build --no-daemon

      - name: Upload JAR as artifact
        uses: actions/upload-artifact@v4
        with:
          name: AetheriumOverlord-jar
          path: build/libs/*.jar
          if-no-files-found: error

      - name: Publish GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: Aetherium Overlord build ${{ github.run_number }}
          files: build/libs/*.jar
          prerelease: false

--- 9.10 ПОШАГОВАЯ ИНСТРУКЦИЯ (5 шагов) ---
 ШАГ 1. На github.com нажмите "+" → "New repository", название aetherium-overlord, Public (или Private), создайте репозиторий.
 ШАГ 2. Нажмите "uploading an existing file" и перетащите папки src, gradle и файлы build.gradle, gradle.properties, settings.gradle, .gitignore (gradlew/gradlew.bat — если есть). Нажмите "Commit changes" в ветку main.
 ШАГ 3. Создайте workflow-файл: "Add file" → "Create new file", в поле имени введите .github/workflows/build.yml (слеши создадут папки), вставьте текст из пункта 9.9, нажмите "Commit changes".
 ШАГ 4. Откройте вкладку "Actions": сборка стартует автоматически после коммита (или нажмите "Run workflow"). Дождитесь зелёной галочки (обычно 3–8 минут).
 ШАГ 5. Нажмите на завершённый запуск → внизу раздел "Artifacts" → скачайте AetheriumOverlord-jar (zip с .jar). Также jar лежит в разделе "Releases" справа на главной странице репозитория. Положите .jar в папку mods вместе с GeckoLib 4 и Curios (версии из gradle.properties) для Forge 1.20.1.

--- 9.11 ВАЖНЫЕ ЗАМЕЧАНИЯ ---
 • Если сборка падает на версиях библиотек (geckolib_version / curios_version) — откройте страницы модов на Modrinth/CurseForge, выберите версию для Forge 1.20.1 и поправьте строки в gradle.properties. Репозитории Maven в build.gradle уже подключены.
 • Все ресурсы (текстуры, модели блоков, blockstates) нужно создать вручную — код их не генерирует. Без них мод соберётся и запустится, но с "заглушками" вместо текстур.
 • Квесты FTB: FTB Quests — отдельный мод, ставится в сборку игры (не в зависимости компиляции).
 • Разделы 1–2, 5, 8 — рабочий каркас кода; боссы, структуры, Алтарь слияния (BlockEntity + рецепты) и остальные 100+ предметов/блоков нужно добавлять по образцам ModItems/ModBlocks.
 • API GeckoLib 4 может немного отличаться между 4.x версиями: при ошибках компиляции CrystalStaffItem сверьтесь с вики GeckoLib 4 (Items) для вашей версии.
